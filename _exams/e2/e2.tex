\question[10]\mbox{}
\textbf{TRUE or FALSE!}  Write 
\begin{enumerate}[(a)]
\item every inflection point of $f(x)$ is also a critical point of $f(x)$
\vspace{1.2in}
\item the function $f(x) = x^{1/3}$ has no critical points
\vspace{1.2in}
\item if $f(x)$ has three different $x$-intercepts, then $f'(x)$ must have at least two $x$-intercepts
\vspace{1.2in}
\item the function $-\frac{x^2}{1+x^2}$ is even
\vspace{1.2in}
\item if $f''(3) = -2$, then $f(x)$ has a local maximum at $x=3$
\vspace{1.2in}
\end{enumerate}

\newpage
\question[10]\mbox{}
A grain silo has the shape of an inverted pyramid with a square base.
The width of the base is $10$ ft and the height of the silo is $30$ ft.
Grain is poured in at a rate of $10$ cubic feet of grain per minute.
How fast is the level of the grain in the tank rising at the exact moment when grain level is $3$ feet high?\\
\textbf{Hint:} the volume of a pyramid with a square base $b$ and height $h$ is $V = \frac{1}{3}b^2h$

\newpage
\question[10]
Calculate the derivatives of the following functions
\begin{enumerate}[(a)]
\item $x^3\cos(x)$
\vspace{3in}
\item $\frac{x^3+1}{x^2+5}$
\vspace{3in}
\item $\tan(\sqrt{x})$
\end{enumerate}

\newpage
\question[10]
Calculate the derivatives of the following functions
\begin{enumerate}[(a)]
\item $\sin\left(\frac{2x-3}{x^5+1}\right)$
\vspace{4in}
\item $(x^8+1)^3\sec(3x+2)$
\end{enumerate}


\newpage
\question[10]\mbox{}
Consider the equation
$$x^9 + 7x^5 + x^5 + 2x^3 + 3x + 4 = 0.$$
\begin{enumerate}[(a)]
\item  State the Mean Value Theorem
\vspace{2in}
\item  Use the Intermediate Value Theorem to show that the above equation has at least one solution
\vspace{3in}
\item  Use the Mean Value Theorem to explain why the equation has no more than one solution
\end{enumerate}

\newpage
\question[10]\mbox{}
For each of the following, provide an example or explain why it does not exist.
\begin{enumerate}[(a)]
\item  a function with no critical points
\vspace{1.2in}
\item  a function with inflection points at $-1$ and $1$
\vspace{1.2in}
\item  a nonconstant function $f(x)$ whose derivative is always zero
\vspace{1.2in}
\item  a function with a global maximum at $x=1$
\vspace{1.2in}
\item  a critical point of the function $3-|x-3|$
\vspace{1.2in}
\end{enumerate}

\newpage
\question[10]\mbox{}
Find the critical points of each of the following functions.
Classify them as local maxima, local minima, or neither.
\begin{enumerate}[(a)]
\item  $2x^3+9x^2+3x+4$
\vspace{4in}
\item  $9x^{5/3}-25x^{3/5}$
\end{enumerate}
