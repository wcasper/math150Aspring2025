\question[20]\mbox{}
\textbf{TRUE or FALSE!}
If the statement is true, write TRUE (all caps, all spelled out).
Otherwise write FALSE (all caps, all spelled out).
\begin{enumerate}[(a)]
\item $f(x) = |x|$ has a critical point
\vspace{0.6in}
\item $\cos(x)$ is concave down at $x=0$
\vspace{0.6in}
\item if $f'(2)$ exists, then $f(x)$ is continuous at $x=2$
\vspace{0.6in}
\item if $f'(1/2) = 2$ then the derivative of $f(\sin(x))$ at $x=\pi/6$ is  $\sqrt{3}$
\vspace{0.6in}
\item the functions $f(x) = x$ and $g(x) = x^2/x$ are the same
\vspace{0.6in}
\item $\int_0^1 f(x)^2dx = \left(\int_0^1 f(x) dx\right)^2$
\vspace{0.6in}
\item $\frac{d}{dx}\sin(2x) = \cos(2x)$
\vspace{0.6in}
\item if $f'(x) = g'(x)$ then $f(x) = g(x)$
\vspace{0.6in}
\item the function $x^{100}-9x^2 + 1$ has a root in the interval $[0,2]$
\vspace{0.6in}
\item a function can never cross its horizontal asymptote
\end{enumerate}

\question[10]\mbox{}
Calculate the following.  Carefully show you work.
\begin{enumerate}[(a)]
\item $\lim_{x\rightarrow 3}\frac{\sqrt{x+6}-\sqrt{3x}}{x-3}$
\vspace{2in}
\item $\lim_{x\rightarrow 1}\frac{\frac{x+3}{x}-\frac{4}{x}}{x-1}$
\vspace{2in}
\item $\lim_{x\rightarrow 0+} x\ln(x)$
\vspace{2in}
\item $\lim_{x\rightarrow 1+} x^{1/(1-x)}$
\vspace{2in}
\end{enumerate}

\newpage
\question[10]
The following is a graph of the \textbf{derivative} $f'(x)$ of a function $f(x)$.
\vspace{0.5in}
\begin{center}
\includegraphics[width=.7\linewidth]{fig/fig1.png}
\end{center}
\begin{enumerate}[(a)]
\item List all intervals where $f(x)$ is decreasing
\vspace{1in}
\item List all intervals where $f(x)$ is concave up
\vspace{1in}
\item List all critical points of $f(x)$ and classify them as local max, local min or neither
\vspace{1in}
\item What is $\lim_{x\rightarrow 8} \frac{f'(x)}{x-8}$?
\vspace{1in}
\item If $f(0) = 4$, is $f(8)$ positive or negative?
\end{enumerate}

\newpage
\question[10]\mbox{}
\begin{enumerate}
\item  Find all critical points of the function $f(x) = xe^{-x}$
\vspace{3.5in}
\item  If $f(2) = 0.75$ and $f'(2) = -0.25$, use the linearization of $f(x)$ at $x=2$ to estimate $f(1.9)$.
\end{enumerate}


\newpage
\question[10]\mbox{}
Find the derivative of each of the following functions.
\begin{enumerate}[(a)]
\item  $\sin^{-1}(x\cos(x))$
\vspace{2.5in}
\item  $\frac{xe^x}{\sin(x) + \cos(x)}$
\vspace{2.5in}
\item  $\sin(x)^{\cos(x)}$
\end{enumerate}

\newpage
\question[10]
Consider an isosceles triangle with base width of $40$ and height of $60$.

\question[10]
Find the area of the region bounded by the graphs $y=x^2-x-1$ and $y=1-x^2$ in the $x,y$--plane.

\question[10]
Compute each of the following integrals.  Show all work.
Lack of work results in no credit!!
\begin{enumerate}[(a)]
\item $\int_1^8 \frac{3 + \sqrt{t}}{t}dt$
\vspace{2in}
\item $\int\frac{2x}{\sqrt{x^2+1}}dx$
\vspace{2in}
\item $\int \frac{3z + 7}{z^2 + 2z + 10} dz$
\vspace{2in}
\item $\int_0^3 (\sqrt{9-x^2} + x + 1)dx$
\end{enumerate}
